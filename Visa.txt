    3. MONGODB SHELL (mongosh) COMMANDS
    ----------------------------------
    mongosh                     Open MongoDB Shell
    show dbs                    Show databases
    use CollegeDB               Switch to database; it is created on first write
    db                          Display the current database
    show collections            List collections in the current database
    db.createCollection("Students")
                                Create a collection
    db.dropDatabase()           Delete the current database
    db.Students.drop()          Delete the Students collection

    INSERT:
    db.Students.insertOne({ studentId: 101, name: "Aarav", course: "MCA", cgpa: 8.5 })
    db.Students.insertMany([{ name: "Aarav" }, { name: "Diya" }])

    READ:
    db.Students.find()          Retrieve all matching documents
    db.Students.findOne()       Retrieve the first matching document
    db.Students.find({ course: "MCA" })
                                Filter by exact value
    db.Students.find({ cgpa: { $gt: 8 } })
                                Greater than
    db.Students.find({ cgpa: { $gte: 8 } })
                                Greater than or equal
    db.Students.find({ cgpa: { $lt: 8 } })
                                Less than
    db.Students.find({ cgpa: { $lte: 8 } })
                                Less than or equal
    db.Students.find({ course: { $ne: "BCA" } })
                                Not equal
    db.Customers.find({ city: { $in: ["Pune", "Mumbai"] } })
                                Match any listed value
    db.Students.find({}, { name: 1, cgpa: 1, _id: 0 })
                                Projection: include selected fields
    db.Students.find().sort({ cgpa: 1 })
                                Sort ascending
    db.Students.find().sort({ cgpa: -1 })
                                Sort descending
    db.Students.find().limit(5)
                                Return at most five documents
    db.Students.find().skip(5)
                                Skip the first five documents

    UPDATE:
    db.Students.updateOne({ studentId: 101 }, { $set: { city: "Pune" } })
    db.Students.updateMany({ course: "MCA" }, { $set: { semester: 3 } })
    db.Students.updateOne({ studentId: 101 }, { $inc: { cgpa: 1 } })
    db.Products.updateMany({ category: "Electronics" }, { $mul: { price: 1.10 } })
                                Multiply price by 1.10 (increase by 10%)

    DELETE:
    db.Students.deleteOne({ studentId: 101 })
    db.Students.deleteMany({ course: "BCA" })
    db.Products.deleteMany({ stock: 0 })

    ARRAY / NULL / MISSING FIELDS:
    db.Orders.updateOne({ OrderID: 1 }, { $push: { Items: { name: "Tea", price: 20 } } })
                                Add an item to an array
    db.IssueRecords.find({ ReturnDate: null })
                                Matches null and missing ReturnDate fields
    db.IssueRecords.find({ ReturnDate: { $exists: false } })
                                Match documents where the field is missing
    db.Customers.find({ Phone: { $exists: false } })
                                Find customers without a Phone field

    AGGREGATION:
    db.Sales.aggregate([{ $match: { Category: "Electronics" } }])
                                Filter documents in a pipeline
    db.Sales.aggregate([{ $group: { _id: "$Category", total: { $sum: 1 } } }])
                                Group and count documents
    db.Results.aggregate([{ $group: { _id: "$Subject", averageMarks: { $avg: "$Marks" } } }])
                                Average marks by subject
    db.Sales.aggregate([{ $project: { total: { $multiply: ["$Quantity", "$UnitPrice"] } } }])
                                Calculate a field using multiplication
    db.Sales.aggregate([
      { $group: { _id: "$Category", revenue: { $sum: { $multiply: ["$Quantity", "$UnitPrice"] } } } },
      { $sort: { revenue: -1 } },
      { $limit: 3 }
    ])
                                Group revenue, sort highest first, show top three

    4. EXPRESS.JS / NODE.JS CODE CONSTRUCTS
    --------------------------------------
    These are JavaScript statements used inside program files, not terminal commands.

    Create an Express app:
    const express = require("express");
    const app = express();

    Start the server:
    app.listen(3000, () => console.log("Server running on port 3000"));

    GET route:
    app.get("/students", (req, res) => {
      res.json(students);
    });

    POST route:
    app.post("/students", (req, res) => {
      res.status(201).json(req.body);
    });

    PUT route:
    app.put("/students/:id", (req, res) => {
      res.json({ id: req.params.id, data: req.body });
    });

    DELETE route:
    app.delete("/students/:id", (req, res) => {
      res.send("Deleted");
    });

    Read route parameter:
    req.params.id

    Read query parameter:
    req.query.department

    Read request body:
    req.body

    Parse JSON request bodies:
    app.use(express.json());

    Parse HTML form submissions:
    app.use(express.urlencoded({ extended: true }));

    Register middleware:
    app.use((req, res, next) => {
      console.log(req.method, req.url);
      next();
    });

    Send JSON:
    res.json({ message: "Success" });

    Set HTTP status:
    res.status(200).json({ message: "OK" });

    Return not found:
    res.status(404).json({ message: "Not found" });

    Serve static files:
    app.use(express.static("public"));

    Configure EJS:
    app.set("view engine", "ejs");

    Render an EJS page:
    res.render("courses", { courses: courses });

    Forward an error:
    next(new Error("Something went wrong"));

    Error-handler signature:
    app.use((err, req, res, next) => {
      res.status(500).send("Something went wrong");
    });

    6. WEBSOCKET COMMANDS AND TESTING
    ---------------------------------
    Install package:
    npm install ws

    Run WebSocket server:
    node server.js

    WebSocket address:
    ws://localhost:8080

    WebSocket server creation:
    const WebSocket = require("ws");
    const server = new WebSocket.Server({ port: 8080 });

    Listen for a client:
    server.on("connection", (socket) => {});

    Receive a message:
    socket.on("message", (message) => {});

    Send a message:
    socket.send("Message received");

    Handle disconnection:
    socket.on("close", () => {});

    Browser client connection:
    const socket = new WebSocket("ws://localhost:8080");

    Send from browser:
    socket.send("Hello server");

    9. IMPORTANT REMINDERS
    ----------------------
    MongoDB sort: 1 = ascending, -1 = descending.
    MongoDB projection: 1 = include, 0 = exclude. _id is included by default.
    MongoDB $gt = greater than; $gte = greater than or equal.
    MongoDB $lt = less than; $lte = less than or equal.
    MongoDB $set = assign a value; $inc = increment; $mul = multiply.
    MongoDB $push = add an element to an array.
    MongoDB $group = group documents; $sum = sum; $avg = average.
    Express req.params = route parameters.
    Express req.query = query-string parameters.
    Express req.body = parsed request body.
    Call next() to pass control to the next middleware.
    Use express.json() for JSON request bodies.
    Use express.urlencoded({ extended: true }) for HTML form bodies.
    Run Node.js programs with node filename.js.
    Stop a running server with Ctrl + C.
